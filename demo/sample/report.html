Add ML-DSA beside the classical signatureSites: 2

Sign with both and verify both, so old verifiers keep working while new ones require the post-quantum signature. Replacement: ML-DSA-65 beside the classical signature.

from cryptography.exceptions import InvalidSignature
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.asymmetric import ec, mldsa

# Keep the classical key and add a post-quantum one. Sign with both.
classical_key = ec.generate_private_key(ec.SECP256R1())
pq_key = mldsa.MLDSA65PrivateKey.generate()


def sign(message: bytes) -> tuple[bytes, bytes]:
    return (
        classical_key.sign(message, ec.ECDSA(hashes.SHA256())),
        pq_key.sign(message),
    )


def verify(message: bytes, signatures: tuple[bytes, bytes]) -> None:
    classical_sig, pq_sig = signatures
    # Accept only if both verify; each raises InvalidSignature otherwise.
    classical_key.public_key().verify(classical_sig, message, ec.ECDSA(hashes.SHA256()))
    pq_key.public_key().verify(pq_sig, message)


signatures = sign(b"release-1.4.2.tar.gz")
verify(b"release-1.4.2.tar.gz", signatures)
try:
    verify(b"tampered", signatures)
    raise AssertionError("a tampered message must not verify")
except InvalidSignature:
    pass

Move the hash to SHA-256 by handSites: 11

The value is stored, compared or sent, so switching it silently would break existing data. Migrate readers and writers together. Replacement: SHA-256.

import hashlib
import hmac

data, key = b"payload", b"shared secret"

# Before: hashlib.sha1(data) and hmac.new(key, data, hashlib.sha1)
digest = hashlib.sha256(data).hexdigest()
tag = hmac.new(key, data, hashlib.sha256).hexdigest()

# Stored values and peers expect the old length: migrate them together, for
# example by writing both digests during a transition period.
assert len(digest) == 64 and len(tag) == 64

from cryptography.hazmat.primitives.asymmetric import mldsa

# This project implements a classical scheme itself. Its users can only migrate if
# it offers a post-quantum signer too. A small wrapper keeps both in one object.


class HybridSigningKey:
    def __init__(self, classical_key) -> None:  # the library's own signing key
        self.classical = classical_key
        self.post_quantum = mldsa.MLDSA65PrivateKey.generate()

    def sign(self, message: bytes) -> tuple[bytes, bytes]:
        return self.classical.sign(message), self.post_quantum.sign(message)


class _Classical:  # stands in for the library's key class in this example
    def sign(self, message: bytes) -> bytes:
        return b"classical-signature"


classical_sig, pq_sig = HybridSigningKey(_Classical()).sign(b"message")
