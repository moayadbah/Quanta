Add ML-DSA beside the classical signatureSites: 58

Sign with both and verify both, so old verifiers keep working while new ones require the post-quantum signature. Replacement: ML-DSA-65 beside the classical signature.

from cryptography.exceptions import InvalidSignature
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.asymmetric import ec, mldsa

# Keep the classical key and add a post-quantum one. Sign with both.
classical_key = ec.generate_private_key(ec.SECP256R1())
pq_key = mldsa.MLDSA65PrivateKey.generate()


def sign(message: bytes) -> tuple[bytes, bytes]:
    return (
        classical_key.sign(message, ec.ECDSA(hashes.SHA256())),
        pq_key.sign(message),
    )


def verify(message: bytes, signatures: tuple[bytes, bytes]) -> None:
    classical_sig, pq_sig = signatures
    # Accept only if both verify; each raises InvalidSignature otherwise.
    classical_key.public_key().verify(classical_sig, message, ec.ECDSA(hashes.SHA256()))
    pq_key.public_key().verify(pq_sig, message)


signatures = sign(b"release-1.4.2.tar.gz")
verify(b"release-1.4.2.tar.gz", signatures)
try:
    verify(b"tampered", signatures)
    raise AssertionError("a tampered message must not verify")
except InvalidSignature:
    pass

Replace RSA encryption with ML-KEMSites: 2

Encapsulate a key with ML-KEM and encrypt the data with AES-GCM under it. Keep RSA only to read old data. Replacement: ML-KEM-768 with AES-256-GCM.

import os

from cryptography.hazmat.primitives.asymmetric import mlkem
from cryptography.hazmat.primitives.ciphers.aead import AESGCM

# Instead of encrypting with RSA-OAEP, encapsulate a fresh key and use AES-GCM.
recipient = mlkem.MLKEM768PrivateKey.generate()

shared_key, encapsulated = recipient.public_key().encapsulate()
nonce = os.urandom(12)
ciphertext = AESGCM(shared_key).encrypt(nonce, b"account export", None)

# Recipient side: send (encapsulated, nonce, ciphertext).
plaintext = AESGCM(recipient.decapsulate(encapsulated)).decrypt(nonce, ciphertext, None)
assert plaintext == b"account export"

from cryptography.hazmat.primitives.asymmetric import ec, ed25519, rsa


def describe(key) -> str:
    # Quanta could not see which algorithm this key uses. Check it where it is loaded.
    if isinstance(key, (rsa.RSAPrivateKey, rsa.RSAPublicKey)):
